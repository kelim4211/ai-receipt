function getSmartSearchURL(queryStr, brandName = "") { let queryText = (queryStr || "").trim(); let cleanBrand = ""; if (brandName && !brandName.includes("정보없음")) { cleanBrand = brandName.replace(/\(AI 추정\)/g, "").replace(/AI 추정/g, "").replace(/\(추정\)/g, "").trim(); } // [철저한 차단] CU, GS25, 세븐일레븐 등 편의점이나 일반 점포명/지점명이 상호에 포함된 경우 검색어 조합에서 완전히 제외 const excludeStoreKeywords = ["CU", "GS25", "세븐일레븐", "이마트24", "미니스톱", "점", "슈퍼", "마트", "상회", "식당", "구내식당", "편의점"]; const isGeneralStore = excludeStoreKeywords.some(keyword => cleanBrand.includes(keyword)); // 편의점이나 일반 소매점일 경우 상호명(브랜드명)을 빼고 오직 순수 제품명만 검색 쿼리로 사용 let fullQuery = (!isGeneralStore && cleanBrand) ? `${cleanBrand} ${queryText}` : queryText; fullQuery = fullQuery.replace(/\s+/g, " ").trim(); if (!fullQuery) return 'https://search.naver.com/search.naver?query='; return `https://search.naver.com/search.naver?query=${encodeURIComponent(fullQuery)}`; }
